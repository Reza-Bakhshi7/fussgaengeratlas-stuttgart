def english_renderer_label(label):
    if label in EXACT_RENDERER_LABELS:
        return EXACT_RENDERER_LABELS[label]
    result = label
    # Longer phrases first.
    for old, new in (
        ("ohne Pfeil", "without arrow"),
        ("Treppe", "Stairs"),
        ("Fußweg", "Footpath"),
        ("Straße", "Street"),
        ("Radweg", "Cycleway"),
        ("Pfeil", "arrow"),
    ):
        result = result.replace(old, new)
    return result
